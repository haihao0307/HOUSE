  const R=rng(seed*97+family*1009),foot=family===0?rectangleFootprint(seed,state.form,scale):family===1?radialFootprint(seed,hero?12:10,1.38*scale,1.02*scale,state.form,3+Math.round(state.fracture*2)):radialFootprint(seed,hero?14:11,1.60*scale,1.08*scale,state.form,3+Math.round(state.fracture*3));

    }sideR.push(ring);
  }
  for(let i=0;i<n;i++){const j=(i+1)%n;for(let ti=0;ti<sideT.length-1;ti++){

  }}

  return{positions:pos,normals,faceKinds:new Float32Array(arr.kind),indices};
}
function makeIcosphere(subdiv=4){const t=(1+Math.sqrt(5))/2,verts=[[-1,t,0],[1,t,0],[-1,-t,0],[1,-t,0],[0,-1,t],[0,1,t],[0,-1,-t],[0,1,-t],[t,0,-1],[t,0,1],[-t,0,-1],[-t,0,1]].map(norm);let faces=[[0,11,5],[0,5,1],[0,1,7],[0,7,10],[0,10,11],[1,5,9],[5,11,4],[11,10,2],[10,7,6],[7,1,8],[3,9,4],[3,4,2],[3,2,6],[3,6,8],[3,8,9],[4,9,5],[2,4,11],[6,2,10],[8,6,7],[9,8,1]];for(let s=0;s<subdiv;s++){const cache=new Map(),next=[];function mid(a,b){const key=a<b?a+','+b:b+','+a;if(cache.has(key))return cache.get(key);const i=verts.length;verts.push(norm(add(verts[a],verts[b])));cache.set(key,i);return i;}for(const[a,b,c]of faces){const ab=mid(a,b),bc=mid(b,c),ca=mid(c,a);next.push([a,ab,ca],[b,bc,ab],[c,ca,bc],[ab,bc,ca]);}faces=next;}return{verts,faces};}
function makeCobble(seed,state,hero=true,scale=1){const ico=makeIcosphere(hero?4:3),R=rng(seed*311+77),axes=[1.28*scale,.73*scale,.96*scale],chipDirs=[];const chips=state.fracture>.35?1+Math.round(state.fracture*2):0;for(let i=0;i<chips;i++)chipDirs.push({n:norm([R()-.5,R()*.48-.08,R()-.5]),d:(.82+R()*.08)*scale,amount:.62+R()*.24});let positions=ico.verts.map(u=>{const n1=fbm3(u[0]*1.25+3.2,u[1]*1.25-4.1,u[2]*1.25+7.3,seed,4),n2=fbm3(u[0]*3.4-1.4,u[1]*3.4+2.8,u[2]*3.4-5.6,seed+701,3),rad=1+n1*.065*state.form+n2*.016*state.relief;let p=[u[0]*axes[0]*rad,u[1]*axes[1]*rad,u[2]*axes[2]*rad];p[0]+=p[1]*(.055+.055*state.form);p[2]+=p[0]*p[1]*.018;for(const c of chipDirs){const over=dot(p,c.n)-c.d;if(over>0)p=sub(p,muls(c.n,over*c.amount));}return p;});const adjacency=Array.from({length:positions.length},()=>new Set());for(const[a,b,c]of ico.faces){adjacency[a].add(b);adjacency[a].add(c);adjacency[b].add(a);adjacency[b].add(c);adjacency[c].add(a);adjacency[c].add(b);}const smoothAmount=.05+.14*state.edge;for(let it=0;it<3;it++){const next=positions.map((p,i)=>{let avg=[0,0,0];for(const j of adjacency[i])avg=add(avg,positions[j]);avg=muls(avg,1/Math.max(1,adjacency[i].size));return [mix(p[0],avg[0],smoothAmount),mix(p[1],avg[1],smoothAmount*.75),mix(p[2],avg[2],smoothAmount)];});positions=next;}let minY=Math.min(...positions.map(p=>p[1])),floor=minY+.12*scale*(.65+state.edge);positions=positions.map(p=>{if(p[1]>=floor)return p;const t=smoothstepJS(minY,floor,p[1]);return[p[0],mix(minY,p[1],t*t),p[2]];});minY=Math.min(...positions.map(p=>p[1]));positions=positions.map(p=>[p[0],p[1]-minY,p[2]]);const pos=new Float32Array(positions.flat()),idx=new Uint32Array(ico.faces.flat()),nor=calcNormals(pos,idx);return{positions:pos,normals:nor,faceKinds:new Float32Array(positions.length),indices:idx};}
function makeStoneGeometry(family,seed,state,hero=true,scale=1){return family===3?makeCobble(seed,state,hero,scale):makePrismaticStone(family,seed,state,hero,scale);}

function makeBeveledCylinder(radius,height,segments=72,bevel=.08){
  const rings=[{y:0,r:radius-bevel},{y:bevel,r:radius},{y:height-bevel,r:radius},{y:height,r:radius-bevel}],pos=[],idx=[];
  for(const ring of rings)for(let i=0;i<=segments;i++){const a=i/segments*Math.PI*2;pos.push(Math.cos(a)*ring.r,ring.y,Math.sin(a)*ring.r);}
  for(let j=0;j<rings.length-1;j++)for(let i=0;i<segments;i++){const a=j*(segments+1)+i,b=a+1,c=a+(segments+1),d=c+1;idx.push(a,c,b,b,c,d);}
  const top=pos.length/3;pos.push(0,height,0);const bot=pos.length/3;pos.push(0,0,0);
  const tr=(rings.length-1)*(segments+1),br=0;for(let i=0;i<segments;i++){idx.push(top,tr+i,tr+i+1);idx.push(bot,br+i+1,br+i);}
  const P=new Float32Array(pos);return{positions:P,normals:calcNormals(P,idx),indices:new Uint32Array(idx)};
}
