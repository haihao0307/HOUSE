  let d=amp*(macro*.78+meso*.22);
  if(family===2){const band=Math.sin((q[1]+macro*.08)*23.0);d+=band*.018*minH*variation;}
  p=add(p,muls(baseN,d));
  for(const c of cuts){const over=dot(p,c.n)-c.threshold;if(over>0)p=sub(p,muls(c.n,over*c.strength));}
  if(family===1){
    const shear=fbm3(q[0]*.72,q[1]*.72,q[2]*.72,seed+933,3);
    p[0]+=p[1]*.055*variation+p[2]*.018*shear;
  }
  if(family===2){p[0]+=p[1]*.12*variation;}
  return p;
}
function makeBoxStone(family,seed,variation,segments=26,mini=false){
  const baseH=family===0?[1.52,.66,.92]:family===1?[1.32,.88,1.06]:[1.56,.39,1.02];
  const h=mini?baseH.map(v=>v*.47):baseH;
  const round=(family===0?.16:family===1?.11:.075)*(mini?.72:1);
  const pos=[],idx=[],cuts=familyCuts(family,seed,h);
  const faces=[
    {axis:0,sign:1,u:2,v:1,flip:true},{axis:0,sign:-1,u:2,v:1,flip:false},
    {axis:1,sign:1,u:0,v:2,flip:true},{axis:1,sign:-1,u:0,v:2,flip:false},
    {axis:2,sign:1,u:0,v:1,flip:false},{axis:2,sign:-1,u:0,v:1,flip:true}
  ];
  const seg=mini?Math.max(12,Math.floor(segments*.58)):segments;
  for(const face of faces){const start=pos.length/3;
    for(let j=0;j<=seg;j++)for(let i=0;i<=seg;i++){
      const p=[0,0,0];p[face.axis]=face.sign*h[face.axis];p[face.u]=mix(-h[face.u],h[face.u],i/seg);p[face.v]=mix(-h[face.v],h[face.v],j/seg);
      let rp=roundedPoint(p,h,Math.min(round,Math.min(...h)*.42));rp=deformStonePoint(rp,h,family,seed,variation,cuts);
      pos.push(rp[0],rp[1],rp[2]);
    }
    for(let j=0;j<seg;j++)for(let i=0;i<seg;i++){
      const a=start+j*(seg+1)+i,b=a+1,c=a+(seg+1),d=c+1;
      if(face.flip)idx.push(a,c,b,b,c,d);else idx.push(a,b,c,b,d,c);
    }
  }
  const P=new Float32Array(pos);shiftBottomToZero(P);const N=smoothSeamNormals(P,calcNormals(P,idx),family===0?.78:family===1?.46:.30);return{positions:P,normals:N,indices:new Uint32Array(idx)};
}
function makePebble(seed,variation,segments=54,mini=false){
  const lon=mini?28:segments,lat=mini?16:30,h=(mini?[.61,.36,.48]:[1.28,.72,.93]),pos=[],idx=[];
  for(let y=0;y<=lat;y++){
    const th=y/lat*Math.PI,st=Math.sin(th),ct=Math.cos(th);
    for(let x=0;x<=lon;x++){
      const ph=x/lon*Math.PI*2,sp=Math.sin(ph),cp=Math.cos(ph),u=[st*cp,ct,st*sp];
      const m=fbm3(u[0]*1.35+4.1,u[1]*1.35-2.8,u[2]*1.35+7.2,seed,4),d=1+(m*.085*mix(.4,1.15,variation));
      const p=[u[0]*h[0]*d,u[1]*h[1]*d,u[2]*h[2]*d];
      p[0]+=p[1]*.06*variation;pos.push(...p);
    }
  }
  for(let y=0;y<lat;y++)for(let x=0;x<lon;x++){const a=y*(lon+1)+x,b=a+1,c=a+(lon+1),d=c+1;idx.push(a,b,c,b,d,c);}
  const P=new Float32Array(pos);shiftBottomToZero(P);return{positions:P,normals:calcNormals(P,idx),indices:new Uint32Array(idx)};
}
function makeStoneGeometry(family,seed,variation,mini=false){return family===3?makePebble(seed,variation,54,mini):makeBoxStone(family,seed,variation,28,mini);}
function makeBeveledCylinder(radius,height,segments=72,bevel=.08){
  const rings=[{y:0,r:radius-bevel},{y:bevel,r:radius},{y:height-bevel,r:radius},{y:height,r:radius-bevel}],pos=[],idx=[];
  for(const ring of rings)for(let i=0;i<=segments;i++){const a=i/segments*Math.PI*2;pos.push(Math.cos(a)*ring.r,ring.y,Math.sin(a)*ring.r);}
  for(let j=0;j<rings.length-1;j++)for(let i=0;i<segments;i++){const a=j*(segments+1)+i,b=a+1,c=a+(segments+1),d=c+1;idx.push(a,c,b,b,c,d);}
  const top=pos.length/3;pos.push(0,height,0);const bot=pos.length/3;pos.push(0,0,0);
  const tr=(rings.length-1)*(segments+1),br=0;for(let i=0;i<segments;i++){idx.push(top,tr+i,tr+i+1);idx.push(bot,br+i+1,br+i);}
  const P=new Float32Array(pos);return{positions:P,normals:calcNormals(P,idx),indices:new Uint32Array(idx)};
}
function makeAnnulus(inner,outer,segments=96){
  const pos=[],normals=[],idx=[];for(let i=0;i<=segments;i++){const a=i/segments*Math.PI*2,c=Math.cos(a),s=Math.sin(a);pos.push(c*inner,0,s*inner,c*outer,0,s*outer);normals.push(0,1,0,0,1,0);}for(let i=0;i<segments;i++){const a=i*2,b=a+1,c=a+2,d=a+3;idx.push(a,c,b,b,c,d);}return{positions:new Float32Array(pos),normals:new Float32Array(normals),indices:new Uint32Array(idx)};
}
function makeCyclorama(){
  const width=60,xSeg=1,profiles=[];
  profiles.push({z:8,y:-1.56,n:[0,1,0]},{z:-3.45,y:-1.56,n:[0,1,0]});
  const r=2.1,steps=20;for(let i=1;i<=steps;i++){const a=i/steps*Math.PI*.5;profiles.push({z:-3.45-r*Math.sin(a),y:-1.56+r*(1-Math.cos(a)),n:[0,Math.cos(a),Math.sin(a)]});}
  profiles.push({z:-5.55,y:7.5,n:[0,0,1]});
  const pos=[],normals=[],idx=[];for(const p of profiles){pos.push(-width/2,p.y,p.z,width/2,p.y,p.z);normals.push(...p.n,...p.n);}for(let j=0;j<profiles.length-1;j++){const a=j*2,b=a+1,c=a+2,d=a+3;idx.push(a,b,c,b,d,c);}return{positions:new Float32Array(pos),normals:new Float32Array(normals),indices:new Uint32Array(idx)};
