    const breakStrength = kind === 'stone' ? parameters.stoneIrregularity * 0.55 : parameters.edgeBreak;
    const inward = minimum * (0.008 + wear * (0.016 + perlin * 0.024) + breakStrength * Math.pow(cell, 3.2) * 0.060 + parameters.pitting * Math.pow(1 - simplex, 4) * 0.018);
    const length = Math.max(0.001, Math.hypot(point.x, point.y));
    const scale = Math.max(0.68, 1 - inward / length);
    const x = point.x * scale;
    const y = point.y * scale;
    if (index === 0) shape.moveTo(x, y); else shape.lineTo(x, y);
  });
  shape.closePath();
  return shape;
}

function erodedBlockGeometry(width, height, depth, seed, noise, kind = 'brick') {
  const shape = erodedRoundedShape(width, height, seed, noise, kind);
  const minimum = Math.min(width, height);
  const bevel = minimum * (kind === 'stone'
    ? 0.025 + parameters.stoneIrregularity * 0.035
    : 0.002 + parameters.edgeRoundness * 0.060);
  const geometry = new THREE.ExtrudeGeometry(shape, {
    depth,
    steps: 1,
    bevelEnabled: true,
    bevelSegments: 3,
    bevelSize: clamp(bevel, 0.0005, minimum * 0.13),
    bevelThickness: Math.min(depth * 0.16, 0.035),
    curveSegments: 5
  });
  geometry.translate(0, 0, -depth / 2);
  const position = geometry.attributes.position;
  for (let index = 0; index < position.count; index += 1) {
    const x = position.getX(index);
    const y = position.getY(index);
    const z = position.getZ(index);
    const fine = noise.perlin.noise(x * 8.2 + seed * 0.013, y * 8.8 - seed * 0.019, z * 7.4 + seed * 0.003);
    const pit = 1 - worley2(x * 11 + seed * 0.007, y * 11 - seed * 0.005, seed + 17);
    const displacement = (fine * 0.008 + Math.pow(pit, 4) * -0.010 * parameters.pitting) * parameters.microRoughness;
    position.setZ(index, z + displacement);
  }
  position.needsUpdate = true;
  geometry.computeVertexNormals();
  return geometry;
}

function irregularPatchGeometry(width, height, depth, seed, noise, loss, side = 1) {
  const count = 52;
  const shape = new THREE.Shape();
  for (let index = 0; index < count; index += 1) {
    const angle = index / count * Math.PI * 2;
    const x0 = Math.cos(angle);
    const y0 = Math.sin(angle);
    const low = noise.simplex.noise(x0 * 1.8 + seed * 0.01, y0 * 1.8 - seed * 0.012);
    const cell = 1 - worley2(x0 * 3.4 + seed * 0.006, y0 * 3.4, seed);
    const radius = clamp(0.82 + low * 0.13 - Math.pow(cell, 2.4) * loss * parameters.plasterEdgeRoughness * 0.22, 0.45, 1.12);
    const x = x0 * width * 0.5 * radius;
    const y = y0 * height * 0.5 * radius;
    if (index === 0) shape.moveTo(x, y); else shape.lineTo(x, y);
  }
  shape.closePath();
  const geometry = new THREE.ExtrudeGeometry(shape, {
    depth,
    bevelEnabled: true,
    bevelSegments: 2,
    bevelSize: Math.min(0.026, depth * 0.7),
    bevelThickness: Math.min(0.012, depth * 0.32),
    curveSegments: 4
  });
  geometry.translate(0, 0, side > 0 ? -depth / 2 : -depth / 2);
  return geometry;
}

function surfaceField(x, y, noise) {
  const scale = parameters.waveScale;
  const warpX = noise.simplex.noise(x * scale * 1.5 + 7.4, y * scale * 1.5 - 3.2);
  const warpY = noise.simplex.noise(x * scale * 1.5 - 8.1, y * scale * 1.5 + 4.7);
  const wx = x + warpX * parameters.domainWarp * 1.45;
  const wy = y + warpY * parameters.domainWarp * 1.45;
  const low = fbmPerlin(noise.perlin, wx * scale, wy * scale, 0.17, 5);
  const erosion = 1 - worley2(wx * scale * 2.4 + 3.1, wy * scale * 2.4 - 2.5, parameters.seed + 9);
  const rain = noise.simplex.noise(wx * 0.34 + 11, wy * 0.075 - 6) * parameters.rainWash * 0.22;
  const damp = Math.pow(clamp(1 - y / Math.max(parameters.wallHeight, 0.1), 0, 1), 3.2) * parameters.dampRise * -0.11;
  return parameters.largeWeathering * (low * 0.095 - Math.pow(erosion, 2.8) * 0.060) + rain * 0.18 + damp;
}

function applyContinuousSurfaceField(geometry, centerX, centerY, noise, side = 1, strength = 1) {
  geometry.computeBoundingBox();
  const position = geometry.attributes.position;
  const faceZ = side > 0 ? geometry.boundingBox.max.z : geometry.boundingBox.min.z;
  const depth = Math.max(0.001, geometry.boundingBox.max.z - geometry.boundingBox.min.z);
  const centerValue = surfaceField(centerX, centerY, noise);
  for (let index = 0; index < position.count; index += 1) {
    const x = position.getX(index);
    const y = position.getY(index);
    const z = position.getZ(index);
    const faceDistance = Math.abs(faceZ - z);
    const influence = 1 - clamp(faceDistance / Math.max(0.018, depth * 0.48), 0, 1);
    if (influence <= 0) continue;
    const delta = (surfaceField(centerX + x, centerY + y, noise) - centerValue) * strength * influence;
    position.setZ(index, z + delta * side);
  }
  position.needsUpdate = true;
  geometry.computeBoundingBox();
  geometry.computeVertexNormals();
  return geometry;
}

function projectedFaceArea(geometry) {
  const position = geometry.attributes.position;
  const index = geometry.index;
  const triangleCount = index ? index.count / 3 : position.count / 3;
  let doubledArea = 0;
  const vertexIndex = (triangle, corner) => index ? index.getX(triangle * 3 + corner) : triangle * 3 + corner;
  for (let triangle = 0; triangle < triangleCount; triangle += 1) {
    const ia = vertexIndex(triangle, 0);
    const ib = vertexIndex(triangle, 1);
    const ic = vertexIndex(triangle, 2);
    const ax = position.getX(ia); const ay = position.getY(ia);
    const bx = position.getX(ib); const by = position.getY(ib);
    const cx = position.getX(ic); const cy = position.getY(ic);
    doubledArea += Math.abs((bx - ax) * (cy - ay) - (by - ay) * (cx - ax)) * 0.5;
  }
  return doubledArea * 0.5;
}

function sampledSurfaceStats(noise, contract) {
  const values = [];
  let fingerprint = 2166136261;
  for (let row = 0; row <= 18; row += 1) {
    const y = contract.wallHeight * row / 18;
    const half = halfWidthAt(y, contract);
    for (let column = 0; column <= 30; column += 1) {
      const x = lerp(-half, half, column / 30);
      const value = surfaceField(x, y, noise);
      values.push(value);
      const quantized = Math.round((value + 1) * 1e6);
      fingerprint ^= quantized;
      fingerprint = Math.imul(fingerprint, 16777619) >>> 0;
    }
  }
  const mean = values.reduce((sum, value) => sum + value, 0) / values.length;
  const variance = values.reduce((sum, value) => sum + (value - mean) ** 2, 0) / values.length;
  return {
    erosionSampleCount: values.length,
    erosionMean: mean,
    erosionStddev: Math.sqrt(variance),
    erosionMin: Math.min(...values),
    erosionMax: Math.max(...values),
    surfaceFingerprint: fingerprint.toString(16).padStart(8, '0')
  };
}

function makeTexture(kind, noise, seedOffset = 0) {
  const size = 256;
  const colorCanvas = document.createElement('canvas');
  const bumpCanvas = document.createElement('canvas');
  colorCanvas.width = colorCanvas.height = bumpCanvas.width = bumpCanvas.height = size;
  const colorContext = colorCanvas.getContext('2d');
  const bumpContext = bumpCanvas.getContext('2d');
  const colorData = colorContext.createImageData(size, size);
  const bumpData = bumpContext.createImageData(size, size);
  const soilBase = [88 + parameters.soilWarmth * 35, 53 + parameters.soilWarmth * 23, 34 + parameters.soilWarmth * 13];
  const plasterBaseValue = 130 + parameters.plasterGray * 70;
  const stoneBase = [103 + parameters.stoneWarmth * 34, 101 + parameters.stoneWarmth * 25, 91 + parameters.stoneWarmth * 15];
  for (let y = 0; y < size; y += 1) {
    for (let x = 0; x < size; x += 1) {
      const u = x / size;
      const v = y / size;
      const qx = noise.simplex.noise(u * 2.4 + seedOffset, v * 2.4 + 9);
      const qy = noise.simplex.noise(u * 2.4 - 5, v * 2.4 + seedOffset);
      const wu = u + qx * parameters.domainWarp * 0.16;
      const wv = v + qy * parameters.domainWarp * 0.16;
      const macro = fbmPerlin(noise.perlin, wu * 3.4, wv * 3.4, seedOffset * 0.01, 5) * 0.5 + 0.5;
      const mid = noise.simplex.noise(wu * 11 + seedOffset, wv * 10) * 0.5 + 0.5;
      const cell = 1 - worley2(wu * 14, wv * 14, parameters.seed + seedOffset);
      const fine = hash(x, y, seedOffset, parameters.seed);
      const damp = parameters.dampRise * Math.pow(1 - v, 2.6) * (0.55 + macro * 0.45);
      const rain = Math.pow(clamp(noise.simplex.noise(u * 20 + seedOffset * 0.1, v * 0.5) * 0.5 + 0.5, 0, 1), 4) * parameters.rainWash;
      let base;
      let height;
      if (kind === 'adobe' || kind === 'core') {
        base = [...soilBase];
        const darkness = parameters.soilDarkness * (kind === 'core' ? 36 : 24);
        const erosion = parameters.pitting * (cell * 0.55 + (1 - mid) * 0.45);
        base[0] += (macro - 0.5) * 34 - darkness - erosion * 26 - damp * 34 - rain * 20;
        base[1] += (macro - 0.5) * 25 - darkness * 0.72 - erosion * 19 - damp * 29 - rain * 17;
        base[2] += (macro - 0.5) * 17 - darkness * 0.48 - erosion * 12 - damp * 18 - rain * 10;
        height = 0.54 + (macro - 0.5) * 0.24 + (mid - 0.5) * 0.15 - cell * parameters.pitting * 0.15 + (fine - 0.5) * 0.055 * parameters.microRoughness;
