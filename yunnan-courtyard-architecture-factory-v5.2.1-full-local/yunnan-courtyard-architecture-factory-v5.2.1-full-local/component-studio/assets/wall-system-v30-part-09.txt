  document.getElementById('toggleWire').addEventListener('click', (event) => {
    wireVisible = !wireVisible;
    event.currentTarget.classList.toggle('active', wireVisible);
    applyVisibility();
  });
}

function revokeReferenceUrl() {
  if (referenceUrl) URL.revokeObjectURL(referenceUrl);
  referenceUrl = '';
}

function showReferenceBlob(blob, label = '墙体参考') {
  revokeReferenceUrl();
  referenceUrl = URL.createObjectURL(blob);
  const image = document.getElementById('referenceImage');
  const placeholder = document.getElementById('referencePlaceholder');
  image.onload = () => {
    image.hidden = false;
    placeholder.hidden = true;
  };
  image.src = referenceUrl;
  image.alt = label;
}

async function listWallReferences() {
  const storage = window.__YUNNAN_COMPONENT_STUDIO_STORAGE__;
  if (!storage) return [];
  const database = await storage.openAttachments();
  return new Promise((resolve, reject) => {
    const store = database.transaction('attachments', 'readonly').objectStore('attachments');
    const request = store.indexNames.contains('moduleId')
      ? store.index('moduleId').getAll(IDBKeyRange.only('walls'))
      : store.getAll();
    request.onsuccess = () => resolve((request.result || []).filter((record) => record.moduleId === 'walls' && (String(record.type || '').startsWith('image/') || /\.(png|jpe?g|webp)$/i.test(record.name || ''))));
    request.onerror = () => reject(request.error || new Error('参考资料读取失败'));
  });
}

function showReferenceAt(index) {
  if (!referenceRecords.length) return;
  referenceIndex = (index + referenceRecords.length) % referenceRecords.length;
  const record = referenceRecords[referenceIndex];
  showReferenceBlob(record.blob, record.name);
  document.getElementById('referenceCounter').textContent = `${referenceIndex + 1} / ${referenceRecords.length}`;
}

async function setupReferences() {
  try {
    referenceRecords = await listWallReferences();
    referenceRecords.sort((a, b) => String(a.createdAt || '').localeCompare(String(b.createdAt || '')));
    if (referenceRecords.length) showReferenceAt(0);
    else document.getElementById('referenceCounter').textContent = '0 / 0';
  } catch (error) {
    document.getElementById('referencePlaceholder').textContent = `本机资料读取失败：${error.message || error}`;
  }
  document.getElementById('referencePrev').addEventListener('click', () => showReferenceAt(referenceIndex - 1));
  document.getElementById('referenceNext').addEventListener('click', () => showReferenceAt(referenceIndex + 1));
  document.getElementById('referenceFile').addEventListener('change', (event) => {
    const file = event.target.files?.[0];
    if (file) showReferenceBlob(file, file.name);
  });
  document.getElementById('referenceFit').addEventListener('click', () => { document.getElementById('referenceImage').style.objectFit = 'contain'; });
  document.getElementById('referenceFill').addEventListener('click', () => { document.getElementById('referenceImage').style.objectFit = 'cover'; });
}

function resize() {
  const width = canvas.clientWidth;
  const height = canvas.clientHeight;
  renderer.setSize(width, height, false);
  camera.aspect = width / Math.max(1, height);
  camera.updateProjectionMatrix();
}

new ResizeObserver(resize).observe(canvas);
function animate() {
  orbit.update();
  renderer.render(scene, camera);
  requestAnimationFrame(animate);
}

window.addEventListener('error', (event) => {
  const errorBox = document.getElementById('canvasError');
  errorBox.textContent = `完整墙体启动失败：${event.message || event.error || '未知错误'}`;
  errorBox.hidden = false;
  document.getElementById('runtimeState').textContent = '运行失败';
  document.getElementById('runtimeState').classList.add('error');
});
window.addEventListener('beforeunload', revokeReferenceUrl);

renderControls();
setupUi();
resize();
rebuild();
setupReferences();
animate();
