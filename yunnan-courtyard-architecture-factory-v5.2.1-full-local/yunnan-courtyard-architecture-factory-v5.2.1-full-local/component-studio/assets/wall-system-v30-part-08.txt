    const node = document.getElementById(id);
    if (!node) return;
    node.textContent = text;
    node.classList.toggle('ok', passed === true);
    node.classList.toggle('error', passed === false);
  };
  set('auditCore', `${geometry.corePieceCount} 个实体 · ${geometry.openSurfaceCount} 开边`, geometry.corePieceCount === 3 && geometry.openSurfaceCount === 0);
  set('auditDoor', contract.closurePassed ? '门框与墙心联动通过' : '门洞闭合失败', contract.closurePassed);
  set('auditTop', `实测 ${geometry.topGapActual.toFixed(3)} m`, geometry.topGapActual <= Math.max(0.030, parameters.topClosureGap + 0.006));
  set('auditJoint', geometry.jointGapActual <= 0.002 ? `实测压接 ${geometry.jointOverlapActual.toFixed(3)} m` : `实测最大空缝 ${geometry.jointGapActual.toFixed(3)} m`, geometry.jointGapActual <= 0.002 && geometry.jointSampleCount >= 20);
  set('auditPier', `${contract.minPierActual.toFixed(2)} m`, contract.minPierActual >= parameters.minPierWidth - 0.001);
  set('auditOpen', String(geometry.openSurfaceCount), geometry.openSurfaceCount === 0);
  set('auditOuterPlaster', `${(geometry.outerPlasterCoverageActual * 100).toFixed(1)}% · ${geometry.outerPatchCount} 片`, geometry.outerPatchCount > 0 && geometry.outerPlasterCoverageActual > 0.05);
  set('auditInnerPlaster', `${(geometry.innerPlasterCoverageActual * 100).toFixed(1)}% · ${geometry.innerPatchCount} 片`, geometry.innerPatchCount > 0 && geometry.innerPlasterCoverageActual > 0.05);
  set('auditErosion', `${geometry.erosionStddev.toFixed(4)} · ${geometry.surfaceFingerprint}`, geometry.erosionSampleCount >= 500 && geometry.erosionStddev > 0.001);
  set('auditSeed', String(parameters.seed), true);
  const jointPassed = geometry.jointGapActual <= 0.002 && geometry.jointSampleCount >= 20;
  const plasterPassed = geometry.outerPatchCount > 0 && geometry.innerPatchCount > 0
    && geometry.outerPlasterCoverageActual > 0.05 && geometry.innerPlasterCoverageActual > 0.05;
  set('gateJoint', jointPassed ? `候选 · 最大缝 ${geometry.jointGapActual.toFixed(3)} m` : '需修改', jointPassed);
  set('gatePlaster', plasterPassed ? `候选 · 外/内 ${(geometry.outerPlasterCoverageActual * 100).toFixed(0)}%/${(geometry.innerPlasterCoverageActual * 100).toFixed(0)}%` : '需修改', plasterPassed);
  set('gateRounded', `几何半径 ${geometry.brickRoundnessActual.toFixed(3)} m`, geometry.brickRoundnessActual > 0.006);
  set('gateErosion', `连续场 ${geometry.surfaceFingerprint}`, geometry.erosionStddev > 0.001);
  set('gateRandom', `Seed ${parameters.seed} · 可复现`, Boolean(geometry.surfaceFingerprint));
}

function applyVisibility() {
  exteriorBrickGroup.visible = !coreReview;
  interiorBrickGroup.visible = !coreReview;
  stoneGroup.visible = !coreReview;
  detailGroup.visible = !coreReview;
  plasterGroup.visible = !coreReview && plasterVisible;
  frameGroup.visible = true;
  guideGroup.visible = wireVisible;
}

function setLayerState(values = {}) {
  if (typeof values.coreReview === 'boolean') coreReview = values.coreReview;
  if (typeof values.plasterVisible === 'boolean') plasterVisible = values.plasterVisible;
  if (typeof values.wireVisible === 'boolean') wireVisible = values.wireVisible;
  document.getElementById('toggleCore')?.classList.toggle('active', coreReview);
  document.getElementById('togglePlaster')?.classList.toggle('active', !plasterVisible);
  document.getElementById('toggleWire')?.classList.toggle('active', wireVisible);
  applyVisibility();
  const state = { coreReview, plasterVisible, wireVisible };
  if (window.__YUNNAN_WALL_SYSTEM_V30__) window.__YUNNAN_WALL_SYSTEM_V30__.layers = { ...state };
  return state;
}

function rebuild() {
  disposeRoot();
  const noise = noiseStack();
  const contract = deriveContract(parameters);
  const materials = makeMaterials(noise);
  buildCore(contract, materials);
  const stones = buildStones(contract, materials, noise);
  const bricks = buildBricks(contract, materials, noise);
  const plaster = buildPlaster(contract, materials, noise);
  buildFrame(contract, materials, noise);
  const details = buildDetails(contract, materials, noise);
  const topology = measureCoreTopology();
  const joint = measureStoneAdobeJoint(contract, stones.stoneRows, bricks.brickRows);
  const erosion = sampledSurfaceStats(noise, contract);
  const geometry = {
    ...stones,
    ...bricks,
    ...plaster,
    ...details,
    ...topology,
    ...joint,
    ...erosion,
    doorClosurePassed: contract.closurePassed,
    brickProjection: parameters.brickProjection,
    innerBrickProjection: parameters.innerBrickProjection,
    brickRoundnessActual: Math.min(parameters.brickLength, parameters.brickHeight) * parameters.edgeRoundness * 0.20,
    measurementMethod: 'world-space-box-sampling-and-indexed-edge-topology'
  };
  geometryRevision += 1;
  lastContract = contract;
  lastGeometry = geometry;
  updateAudit(contract, geometry);
  updateMetrics(geometry);
  applyVisibility();
  document.body.dataset.wallSystemReady = 'true';
  window.__YUNNAN_WALL_SYSTEM_V30__ = {
    version: VERSION,
    parameters: { ...parameters },
    contract: { ...contract },
    geometry: { ...geometry },
    revision: geometryRevision,
    view: activeView,
    layers: { coreReview, plasterVisible, wireVisible },
    controlDefs,
    presets,
    applyParameters(values) {
      parameters = { ...parameters, ...values };
      renderControls();
      rebuild();
    },
    rebuild,
    setView,
    setLayerState
  };
}

function formatValue(value, step) {
  if (step >= 1) return String(Math.round(value));
  if (step < 0.005) return Number(value).toFixed(3);
  return Number(value).toFixed(2);
}

function renderControls() {
  let count = 0;
  for (const [hostId, definitions] of Object.entries(controlDefs)) {
    const host = document.getElementById(hostId);
    if (!host) continue;
    const nodes = definitions.map((definition) => {
      count += 1;
      const wrapper = document.createElement('div');
      wrapper.className = 'control';
      const label = document.createElement('label');
      const text = document.createElement('span');
      text.textContent = definition.label;
      const output = document.createElement('output');
      output.textContent = formatValue(parameters[definition.key], definition.step);
      label.append(text, output);
      const input = document.createElement('input');
      input.type = 'range';
      input.min = definition.min;
      input.max = definition.max;
      input.step = definition.step;
      input.value = parameters[definition.key];
      input.dataset.parameter = definition.key;
      input.addEventListener('input', () => {
        parameters[definition.key] = Number(input.value);
        output.textContent = formatValue(parameters[definition.key], definition.step);
        clearTimeout(rebuildTimer);
        rebuildTimer = setTimeout(rebuild, 95);
      });
      wrapper.append(label, input);
      return wrapper;
    });
    host.replaceChildren(...nodes);
  }
  document.getElementById('controlCount').textContent = `${count} 项`;
}

function setView(name) {
  const views = {
    oblique: { position: [9.7, 4.8, 11.6], target: [0, 2.45, 0], fov: 38 },
    front: { position: [0, 2.75, 12.4], target: [0, 2.45, 0], fov: 38 },
    interior: { position: [-7.8, 4.0, -10.4], target: [0, 2.45, 0], fov: 38 },
    plaster: { position: [-1.8, 3.8, 8.0], target: [-0.9, 3.15, 0], fov: 34 },
    joint: { position: [4.7, 1.65, 4.2], target: [2.65, 1.03, 0], fov: 30 },
    close: { position: [4.15, 3.55, 4.5], target: [2.25, 3.15, 0], fov: 28 }
  };
  const view = views[name] || views.oblique;
  activeView = views[name] ? name : 'oblique';
  camera.position.fromArray(view.position);
  camera.fov = view.fov;
  camera.updateProjectionMatrix();
  orbit.target.fromArray(view.target);
  orbit.update();
  document.querySelectorAll('[data-view]').forEach((button) => button.classList.toggle('active', button.dataset.view === activeView));
  if (window.__YUNNAN_WALL_SYSTEM_V30__) window.__YUNNAN_WALL_SYSTEM_V30__.view = activeView;
  return activeView;
}

function setupUi() {
  document.querySelectorAll('[data-preset]').forEach((button) => {
    button.addEventListener('click', () => {
      parameters = { ...presets[button.dataset.preset], seed: parameters.seed };
      document.querySelectorAll('[data-preset]').forEach((item) => item.classList.toggle('active', item === button));
      renderControls();
      rebuild();
    });
  });
  document.querySelectorAll('[data-view]').forEach((button) => button.addEventListener('click', () => setView(button.dataset.view)));
  document.getElementById('toggleCore').addEventListener('click', (event) => {
    setLayerState({ coreReview: !coreReview });
  });
  document.getElementById('togglePlaster').addEventListener('click', (event) => {
    setLayerState({ plasterVisible: !plasterVisible });
  });
